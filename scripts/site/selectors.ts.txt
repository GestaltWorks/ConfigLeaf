interface TraceEvent {
  agent: string;
  action: string;
  status: string;
  input_summary?: string;
  output_summary?: string;
  detail?: Record<string, unknown>;
}

function formatTraceLine(e: TraceEvent, redact: (s: string) => string): string {
  const parts = [
    `${e.agent}/${e.action}`,
    e.status,
    e.input_summary,
    e.output_summary,
  ].filter(Boolean);
  return redact(parts.join(" · "));
}

function selectActTwoSummaries(
  events: TraceEvent[],
  redact: (s: string) => string
): string {
  return events
    .filter(
      (e) =>
        (e.agent === "detector" &&
          e.action === "detect" &&
          e.output_summary?.startsWith("found_but_different")) ||
        (e.agent === "actor" && e.action === "revert_record")
    )
    .map((e) => formatTraceLine(e, redact))
    .join("\n");
}

function selectActThreeSummaries(
  events: TraceEvent[],
  redact: (s: string) => string
): string {
  return events
    .filter(
      (e) =>
        e.agent === "actor" &&
        (e.action === "investigate" || e.action === "fix")
    )
    .map((e) => formatTraceLine(e, redact))
    .join("\n");
}

function selectLabChainSummaries(
  events: TraceEvent[],
  redact: (s: string) => string
): string {
  return events
    .filter((e) => e.agent === "explainer" && e.action === "explain")
    .map((e) => formatTraceLine(e, redact))
    .join("\n");
}
